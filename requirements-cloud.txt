# Cloud deployment dependencies (Hugging Face Spaces / any CPU host).
#
# Generated from requirements.txt with the local-GPU stack removed:
#   torch, diffusers, transformers, accelerate, safetensors
# Those exist only for local image generation via imagegen.py, which
# needs a GPU to be usable. They add ~2.3GB to the image for a feature
# that would time out on 2 shared vCPUs, so the cloud build uses a
# hosted image API instead.

Flask==3.0.3
python-dotenv==1.0.1
requests==2.32.3
stripe==11.1.1
pypdf==6.16.1
python-docx==1.2.0

# Production WSGI server. Flask's built-in server is single-process and
# explicitly not for public traffic; Render runs this instead.
gunicorn==23.0.0

# Symbolic maths, for the code sandbox. codeexec.py grants this one
# package by name (see the note there) so generated maths can actually be
# computed rather than reasoned about - which means it has to be present
# on the server, not only on a developer machine. ~40MB with mpmath, and
# pure Python, so it adds no build step on ARM.
sympy==1.14.0

# Resampling only, for imagegen.py. The keyless image service caps its
# output at ~0.59MP whatever size is requested, so a 1024x1024 request
# arrives as 768x768; _fit() scales it to the size the app promised.
# Pure wheels on ARM, so it adds no build step on the Oracle VM.
Pillow==12.3.0

# Free-tier text-to-video (hfvideo.py). Pure Python, no torch - the
# client only speaks HTTP to Hugging Face, which routes to whichever
# provider is serving the model. Optional: without HF_TOKEN set,
# hfvideo.configured() is False and the bay simply says so.
huggingface_hub>=0.34
